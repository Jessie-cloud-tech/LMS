cmake_minimum_required(VERSION 3.16)
project(ModeratedLMS LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

include(FetchContent)

# SQLite (downloaded and compiled automatically)
FetchContent_Declare(sqlite_amalgamation
  URL https://www.sqlite.org/2024/sqlite-amalgamation-3460000.zip)
FetchContent_MakeAvailable(sqlite_amalgamation)
add_library(sqlite3_lib STATIC ${sqlite_amalgamation_SOURCE_DIR}/sqlite3.c)
target_include_directories(sqlite3_lib PUBLIC ${sqlite_amalgamation_SOURCE_DIR})
find_package(Threads REQUIRED)
target_link_libraries(sqlite3_lib PUBLIC Threads::Threads ${CMAKE_DL_LIBS})

# libsodium (password hashing)
set(SODIUM_DISABLE_TESTS ON CACHE BOOL "" FORCE)
FetchContent_Declare(sodium
  GIT_REPOSITORY https://github.com/robinlinden/libsodium-cmake.git
  GIT_TAG master)
FetchContent_MakeAvailable(sodium)

file(GLOB LMS_SOURCES CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/src/*.cpp ${CMAKE_SOURCE_DIR}/main.cpp)
add_executable(lms ${LMS_SOURCES})
target_include_directories(lms PRIVATE include)
target_link_libraries(lms PRIVATE sqlite3_lib sodium)
if(MSVC)
  target_compile_definitions(lms PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
